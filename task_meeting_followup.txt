import json
from typing import Any
from openai import OpenAI
from app.config import settings
from app.models import WorkflowState

_client = OpenAI(api_key=settings.openai_api_key)

def draft_followup(state: WorkflowState) -> dict:
    """For each action item, generate a short professional follow-up message."""
    drafts = []
    for item in state.get("action_items", []):
        prompt = (
            f"Write a short (2-3 sentence) professional follow-up message to "
            f"{item['owner']} about the action item: {item['task']}. "
            f"Due: {item.get('due_date') or 'no specific date'}. "
            f"Tone: polite, direct. No subject line, just the message body."
        )
        resp = _client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3,
        )
        drafts.append({
            "owner": item["owner"],
            "task": item["task"],
            "message": resp.choices[0].message.content.strip(),
        })
    return {"drafts": drafts, "status": "pending_approval"}

def wait_for_approval(state: WorkflowState) -> dict:
    """No-op node — the interrupt is placed BEFORE this node in the graph.
    When the graph resumes after approval, control returns here and then
    proceeds to mark_sent. Return state unchanged."""
    return state

def mark_sent(state: WorkflowState) -> dict:
    return {"status": "sent"}

def mark_rejected(state: WorkflowState) -> dict:
    return {"status": "rejected"}

Build the graph with Postgres checkpointing.

import psycopg
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres import PostgresSaver

from app.config import settings
from app.models import WorkflowState
from app.graph.nodes import (
    draft_followup, wait_for_approval, mark_sent, mark_rejected,
)

# IMPORTANT — psycopg connection with autocommit for the checkpointer.
# The PostgresSaver needs the connection in autocommit mode.
_conn = psycopg.connect(settings.checkpoint_db_url, autocommit=True)
checkpointer = PostgresSaver(_conn)

# CRITICAL: call setup() exactly once before any workflow runs.
# Without this, the checkpoint tables don't exist and every workflow fails.
checkpointer.setup()

def _route_after_wait(state: WorkflowState) -> str:
    """Route based on the decision recorded by the API when resuming."""
    decision = state.get("decision")
    if decision == "approve":
        return "approve"
    if decision == "reject":
        return "reject"
    return "approve"  # default to approve if no decision set

def build_graph():
    builder = StateGraph(WorkflowState)
    builder.add_node("draft_followup", draft_followup)
    builder.add_node("wait_for_approval", wait_for_approval)
    builder.add_node("mark_sent", mark_sent)
    builder.add_node("mark_rejected", mark_rejected)

    builder.add_edge(START, "draft_followup")
    builder.add_edge("draft_followup", "wait_for_approval")

    builder.add_conditional_edges(
        "wait_for_approval",
        _route_after_wait,
        {"approve": "mark_sent", "reject": "mark_rejected"},
    )

    builder.add_edge("mark_sent", END)
    builder.add_edge("mark_rejected", END)

    # interrupt_before pauses the graph BEFORE running wait_for_approval.
    # So draft_followup runs → graph pauses → API returns pending_approval.
    # On resume, wait_for_approval runs → conditional routes to mark_sent/rejected.
    return builder.compile(
        checkpointer=checkpointer,
        interrupt_before=["wait_for_approval"],
    )

graph = build_graph()

from uuid import UUID
from datetime import datetime
from fastapi import APIRouter, Depends, HTTPException, Request
from slowapi import Limiter
from slowapi.util import get_remote_address
from sqlalchemy.orm import Session

from app.db import get_db
from app.models import (
    StartWorkflowRequest, WorkflowOut, WorkflowListItem, Workflow, WorkflowState,
)
from app.graph.workflow import graph

limiter = Limiter(key_func=get_remote_address)
router = APIRouter(tags=["workflows"])

@router.post("/workflows", response_model=WorkflowOut)
@limiter.limit("10/minute")
def start_workflow(
    request: Request,
    req: StartWorkflowRequest,
    db: Session = Depends(get_db),
):
    workflow_id = __import__("uuid").uuid4()

    # Persist a row first so GET can find it even if graph execution fails
    wf = Workflow(
        id=workflow_id,
        meeting_id=req.meeting_id,
        status="drafting",
        drafts=[],
    )
    db.add(wf); db.commit(); db.refresh(wf)

    # Run the graph up to the interrupt
    initial_state: WorkflowState = {
        "meeting_id": str(req.meeting_id),
        "action_items": [a.model_dump() for a in req.action_items],
        "drafts": [],
        "decision": None,
        "status": "drafting",
    }
    config = {"configurable": {"thread_id": str(workflow_id)}}

    try:
        graph.invoke(initial_state, config=config)
    except Exception as e:
        db.delete(wf); db.commit()
        raise HTTPException(status_code=500, detail=f"Graph failed: {e}")

    # After invoke returns, the graph has paused at wait_for_approval
    state = graph.get_state(config)
    drafts = state.values.get("drafts", [])
    wf.drafts = drafts
    wf.status = "pending_approval"
    db.commit(); db.refresh(wf)

    return WorkflowOut.model_validate(wf)

@router.get("/workflows", response_model=list[WorkflowListItem])
def list_workflows(db: Session = Depends(get_db)):
    return db.query(Workflow).order_by(Workflow.created_at.desc()).all()

@router.get("/workflows/{workflow_id}", response_model=WorkflowOut)
def get_workflow(workflow_id: UUID, db: Session = Depends(get_db)):
    wf = db.query(Workflow).filter(Workflow.id == workflow_id).first()
    if not wf:
        raise HTTPException(status_code=404, detail="Workflow not found")
    return WorkflowOut.model_validate(wf)

@router.post("/workflows/{workflow_id}/approve", response_model=WorkflowOut)
@limiter.limit("10/minute")
def approve_workflow(
    request: Request,
    workflow_id: UUID,
    db: Session = Depends(get_db),
):
    wf = db.query(Workflow).filter(Workflow.id == workflow_id).first()
    if not wf:
        raise HTTPException(status_code=404, detail="Workflow not found")
    if wf.status != "pending_approval":
        raise HTTPException(status_code=400, detail=f"Cannot approve in status {wf.status}")

    config = {"configurable": {"thread_id": str(workflow_id)}}
    graph.update_state(config, {"decision": "approve"})
    graph.invoke(None, config=config)   # resume from interrupt

    wf.status = "sent"
    wf.decided_at = datetime.utcnow()
    db.commit(); db.refresh(wf)
    return WorkflowOut.model_validate(wf)

@router.post("/workflows/{workflow_id}/reject", response_model=WorkflowOut)
@limiter.limit("10/minute")
def reject_workflow(
    request: Request,
    workflow_id: UUID,
    db: Session = Depends(get_db),
):
    wf = db.query(Workflow).filter(Workflow.id == workflow_id).first()
    if not wf:
        raise HTTPException(status_code=404, detail="Workflow not found")
    if wf.status != "pending_approval":
        raise HTTPException(status_code=400, detail=f"Cannot reject in status {wf.status}")

    config = {"configurable": {"thread_id": str(workflow_id)}}
    graph.update_state(config, {"decision": "reject"})
    graph.invoke(None, config=config)

    wf.status = "rejected"
    wf.decided_at = datetime.utcnow()
    db.commit(); db.refresh(wf)
    return WorkflowOut.model_validate(wf)

If the exact API for resuming from an interrupt differs in your installed
langgraph version, inspect `help(graph.update_state)` and `help(graph.invoke)`
and adapt. The pattern above is the documented LangGraph HITL pattern for
versions 0.2+ — update_state sets state, invoke(None) resumes.
